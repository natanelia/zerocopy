import { describe, expect, it } from 'vitest';
import { Arena, HEAP_START, arenaOf, vectorDepth } from './arena';
import { SharedList, getWorkerData, initWorker } from './shared';

function fresh<T extends string>(type: T, values: readonly any[], arena = new Arena()) {
  return new SharedList<T>(type, 0, 0, 0, arena).pushMany(values);
}
function address(list: SharedList<any>, index: number) {
  const start = (list.size - 1) & ~31, a = arenaOf(list);
  return index >= start ? list.tail + (index - start) * 8
    : (a.wasm.vecLeaf(list.root, list.depth, index) >>> 0) + (index & 31) * 8;
}

// Import one existing (i32,i32,i32,f64)->i32 WASM export and expose an i64
// payload wrapper. Reinterpretation stays inside WASM: JavaScript NaN boxing
// cannot silently turn distinct test payloads into one canonical number.
const bitBridge = new WebAssembly.Module(new Uint8Array([
  0, 97, 115, 109, 1, 0, 0, 0,
  1, 17, 2, 96, 4, 127, 127, 127, 124, 1, 127, 96, 4, 127, 127, 127, 126, 1, 127,
  2, 11, 1, 3, 101, 110, 118, 3, 115, 101, 116, 0, 0,
  3, 2, 1, 1,
  7, 11, 1, 7, 115, 101, 116, 66, 105, 116, 115, 0, 1,
  10, 15, 1, 13, 0, 32, 0, 32, 1, 32, 2, 32, 3, 191, 16, 0, 11,
]));
function exactSetter(a: Arena, tail: boolean) {
  const instance = new WebAssembly.Instance(bitBridge, { env: { set: tail ? a.wasm.tailSet : a.wasm.vecSet } });
  return instance.exports.setBits as (root: number, depthOrLength: number, index: number, bits: bigint) => number;
}
function rawFixture(size: number, bits: bigint) {
  const a = new Arena(), input = a.alloc(size * 8), view = a.dv;
  for (let i = 0; i < size; i++) view.setBigUint64(input + i * 8, bits, true);
  const length = ((size - 1) & 31) + 1, prefix = size - length, depth = vectorDepth(prefix);
  const root = a.wasm.vecLink(0, 0, depth, 0, input, prefix) >>> 0;
  // Every byte is initialized before publishing this valid list descriptor.
  return new SharedList('number', root, depth, size, a, input + prefix * 8);
}

describe('list set with an exact-bit tree lookup before copying', () => {
  for (const size of [1, 31, 32, 33, 65, 1025, 32769, 65537]) {
    it(`reuses only existing tree storage while preserving tail copies at size ${size}`, () => {
      const values = Array.from({ length: size }, (_, i) => i + .25), list = fresh('number', values);
      const a = arenaOf(list), used = a.used, bytes = a.buf.slice(HEAP_START, used), descriptor = list.toWorkerData();
      const indices = [...new Set([0, Math.min(31, size - 1), Math.max(0, size - 2), size - 1])];
      for (const index of indices) {
        const mark = a.used, tailStart = (size - 1) & ~31, inTree = index < tailStart;
        const next = list.set(index, values[index]);
        expect(next).not.toBe(list); expect(Object.isFrozen(next)).toBe(true);
        expect(arenaOf(next)).toBe(a); expect(next.root).toBe(list.root);
        if (inTree) expect(next.toWorkerData()).toEqual(descriptor);
        else expect(next.tail).not.toBe(list.tail);
        expect(next.toArray()).toEqual(values); expect(a.used - mark).toBe(inTree ? 0 : (size - tailStart) * 8);
      }
      expect(list.toArray()).toEqual(values); expect(a.buf.slice(HEAP_START, used)).toEqual(bytes);
    });
  }

  it('keeps forks and frontier appends independent after no-op writes', () => {
    const values = Array.from({ length: 33 }, (_, i) => i), original = fresh('number', values);
    const noop = original.set(0, 0), left = noop.push(101), right = original.push(202);
    const changed = noop.set(0, -1), grow = changed.pushMany(Array.from({ length: 4096 }, (_, i) => 1000 + i));
    expect(noop.toArray()).toEqual(values); expect(original.toArray()).toEqual(values);
    expect(left.toArray()).toEqual([...values, 101]); expect(right.toArray()).toEqual([...values, 202]);
    expect(changed.toArray()).toEqual([-1, ...values.slice(1)]); expect(grow.size).toBe(4129);
  });

  it('reuses boolean and interned-string tree storage but still encodes generic values', () => {
    for (const [type, values] of [
      ['boolean', Array.from({ length: 65 }, (_, i) => !!(i & 1))],
      ['string', Array.from({ length: 65 }, (_, i) => `界🙂-${i % 4}`)],
    ] as const) {
      const list = fresh(type, values), a = arenaOf(list), used = a.used;
      for (const index of [0, 31, 32, 63]) expect(list.set(index, values[index] as never).toWorkerData()).toEqual(list.toWorkerData());
      expect(a.used).toBe(used);
      const tail = list.set(64, values[64] as never);
      expect(tail.tail).not.toBe(list.tail); expect(a.used - used).toBe(8);
    }
    const a = new Arena(), list = fresh('object', Array.from({ length: 33 }, () => ({ old: 1 })), a), used = a.used;
    let calls = 0;
    const value = { toJSON() { calls++; a.memory.grow(1); return { old: 1 }; } };
    const next = list.set(0, value);
    expect(calls).toBe(1); expect(a.used).toBeGreaterThan(used); expect(next.root).not.toBe(list.root);
    expect(next.get(0)).toEqual({ old: 1 }); expect(list.get(0)).toEqual({ old: 1 });
    expect(Object.isFrozen(next.get(0))).toBe(true);
    const child = fresh('number', [1, 2, 3]), nested = fresh('SharedList<number>', Array.from({ length: 33 }, () => child), a), mark = a.used;
    const nestedNext = nested.set(0, child);
    expect(a.used).toBeGreaterThan(mark); expect(nestedNext.get(0)!.toArray()).toEqual([1, 2, 3]);
    expect(nested.get(0)!.toArray()).toEqual([1, 2, 3]);
  });

  it('keeps validation and invalid-index identity before reaching WASM', () => {
    const values = Array.from({ length: 33 }, (_, i) => i + .25); values[0] = 0; values[1] = -0; values[2] = NaN;
    const list = fresh('number', values), a = arenaOf(list), used = a.used;
    for (const index of [-1, 33, .5, NaN, Infinity]) expect(list.set(index, 'wrong' as never)).toBe(list);
    expect(() => list.set(0, 'wrong' as never)).toThrow(/Expected a number/);
    expect(a.used).toBe(used);
    for (const [index, value] of [[0, -0], [1, 0], [2, Infinity]] as const) {
      const next = list.set(index, value);
      expect(Object.is(next.get(index), value)).toBe(true); expect(next.root).not.toBe(list.root);
    }
    expect(Object.is(list.get(0), 0)).toBe(true); expect(Object.is(list.get(1), -0)).toBe(true); expect(Number.isNaN(list.get(2))).toBe(true);
  });

  for (const copy of [false, true]) it(`still rejects same-value writes in a read-only reader (copy=${copy})`, async () => {
    const list = fresh('number', Array.from({ length: 65 }, (_, i) => i)), { reader } = await initWorker<{ reader: SharedList<'number'> }>(getWorkerData({ reader: list }, { copy }));
    const a = arenaOf(reader), used = a.used, bytes = a.copy();
    for (const index of [0, 64]) expect(() => reader.set(index, index)).toThrow(/read-only/);
    expect(reader.set(-1, 'wrong' as never)).toBe(reader);
    expect(a.used).toBe(used); expect(a.copy()).toEqual(bytes);
    expect(reader.toArray()).toEqual(list.toArray());
  });

  for (const size of [1, 33, 1057]) it(`compares exact signed zero and NaN payload bits at size ${size}`, () => {
    const payloads = [0n, 0x8000000000000000n, 0x7ff8000000000042n, 0x7ff8000000000043n, 0xfff8000000000042n, 0x7ff0000000000001n];
    for (const bits of payloads) {
      const list = rawFixture(size, bits), a = arenaOf(list), used = a.used, bytes = a.buf.slice(HEAP_START, used);
      for (const index of [...new Set([0, size - 1])]) {
        const start = (size - 1) & ~31, tail = index >= start, set = exactSetter(a, tail);
        const old = tail ? list.tail : list.root, lengthOrDepth = tail ? size - start : list.depth, relative = tail ? index - start : index;
        const mark = a.used;
        const equal = set(old, lengthOrDepth, relative, bits) >>> 0;
        if (tail) { expect(equal).not.toBe(old); expect(a.used - mark).toBe(lengthOrDepth * 8); }
        else { expect(equal).toBe(old); expect(a.used).toBe(mark); }
        const changedBits = bits ^ 1n, root = set(old, lengthOrDepth, relative, changedBits) >>> 0;
        expect(root).not.toBe(old);
        const next = new SharedList('number', tail ? list.root : root, list.depth, size, a, tail ? root : list.tail);
        expect(a.dv.getBigUint64(address(next, index), true)).toBe(changedBits);
        expect(a.dv.getBigUint64(address(list, index), true)).toBe(bits);
        expect(a.used).toBeGreaterThan(mark);
        expect(a.buf.slice(HEAP_START, used)).toEqual(bytes);
      }
    }
  });

  it('still allocates missing zero-valued leaves through the existing WASM ABI', () => {
    const a = new Arena(), used = a.used;
    const root = a.wasm.vecSet(0, 0, 0, 0) >>> 0;
    expect(root).not.toBe(0); expect(a.used - used).toBe(256); expect(a.wasm.vecGet(root, 0, 0)).toBe(0);
    const tail = a.wasm.tailSet(0, 1, 0, 0) >>> 0;
    expect(tail).not.toBe(0); expect(a.dv.getFloat64(tail, true)).toBe(0);
    // A nonzero branch can still contain a missing path. Never read address zero
    // and mistake its zero bits for a published leaf's unchanged value.
    const branch = a.alloc(128); a.buf.fill(0, branch, branch + 128); const mark = a.used;
    const populated = a.wasm.vecSet(branch, 1, 0, 0) >>> 0;
    expect(populated).not.toBe(branch); expect(a.used - mark).toBe(384);
    expect(a.dv.getUint32(branch, true)).toBe(0); expect(a.wasm.vecGet(populated, 1, 0)).toBe(0);
  });

  it('can return an equal snapshot at full capacity while changed writes still fail safely', () => {
    const a = new Arena({ memory: new WebAssembly.Memory({ initial: 2, maximum: 2, shared: true }) });
    const values = Array.from({ length: 65 }, (_, i) => i), list = fresh('number', values, a);
    const published = a.buf.slice(HEAP_START, a.used);
    a.alloc(2 * 65536 - a.used); const used = a.used;
    const next = list.set(0, values[0]); expect(next).not.toBe(list); expect(next.toArray()).toEqual(values);
    expect(a.used).toBe(used); expect(() => list.set(0, -1)).toThrow(WebAssembly.RuntimeError);
    // The tail path is the original copy operation, even for identical bits.
    expect(() => list.set(64, values[64])).toThrow(WebAssembly.RuntimeError);
    expect(() => list.set(64, -1)).toThrow(WebAssembly.RuntimeError);
    expect(list.toArray()).toEqual(values); expect(a.buf.slice(HEAP_START, HEAP_START + published.length)).toEqual(published);
  });
});
